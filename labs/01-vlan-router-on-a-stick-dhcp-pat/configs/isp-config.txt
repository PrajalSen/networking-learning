! Lab 01 - ISP router configuration
! Provides the outside network and fake Internet destination

enable
configure terminal

interface g0/0
 ip address 200.1.1.2 255.255.255.0
 no shutdown
exit

interface loopback0
 ip address 8.8.8.8 255.255.255.255
exit

end
write memory

! Verification:
! show ip interface brief
! ping 200.1.1.1
